```python
def summarize(scores):
    if not scores:
        return None
    total = 0
    highest = scores[0]
    lowest = scores[0]
    passed = 0
    for score in scores:
        total += score
        if score > highest:
            highest = score
        if score < lowest:
            lowest = score
        if score >= 60:
            passed += 1
    return total / len(scores), highest, lowest, passed

n = int(input())
scores = []
for i in range(max(0, n)):
    score = float(input())
    while score < 0 or score > 100:
        print("Invalid input")
        score = float(input())
    scores.append(score)
result = summarize(scores)
if result is None:
    print("暂无成绩")
else:
    average, highest, lowest, passed = result
    print(f"平均分：{average:.2f}")
    print(f"最高分：{highest:.2f}")
    print(f"最低分：{lowest:.2f}")
    print(f"及格人数：{passed}/{len(scores)}")
```

- 计算函数对空列表先返回 None，再进行最值与平均数计算。
- 录入与输出保留在函数外，因此能单独验证计算。
